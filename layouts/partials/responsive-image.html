{{/* Local PNG and JPEG images are converted to responsive WebP images. Other image types and images Hugo cannot find are rendered unchanged. Required params: src and alt. Optional params: page, sizes, class, style, title, and loading. */}} {{/* Resolve local images from page bundles, assets, shared content, or static. */}} {{- $resource := false -}} {{- $url := urls.Parse .src -}} {{- if not (or $url.IsAbs (strings.HasPrefix .src "//")) -}} {{- $resourcePath := strings.TrimPrefix "/" (path.Clean $url.Path) -}} {{- with .page -}} {{- with .Resources.GetMatch $resourcePath -}} {{- $resource = . -}} {{- else -}} {{- with $.page.Resources.GetMatch (path.Base $resourcePath) -}} {{- $resource = . -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $resource -}} {{- $resource = resources.Get $resourcePath -}} {{- end -}} {{- if not $resource -}} {{- with site.GetPage (path.Dir $resourcePath) -}} {{- $resource = .Resources.GetMatch (path.Base $resourcePath) -}} {{- end -}} {{- end -}} {{- if not $resource -}} {{- $staticPath := printf "static/%s" $resourcePath -}} {{- if fileExists $staticPath -}} {{- $resource = resources.FromString $resourcePath (readFile $staticPath) -}} {{- end -}} {{- end -}} {{- end -}} {{/* Convert PNG/JPEG sources to WebP without generating widths larger than the original. */}} {{- $processable := and $resource (or (eq $resource.MediaType.SubType "jpeg") (eq $resource.MediaType.SubType "png")) -}} {{- if $processable -}} {{- $candidates := slice -}} {{- range slice 480 800 1200 -}} {{- if lt . $resource.Width -}} {{- $candidates = $candidates | append ($resource.Resize (printf "%dx webp q82" .)) -}} {{- end -}} {{- end -}} {{- $candidates = $candidates | append ($resource.Process "webp q82") -}} {{- $src := index $candidates (sub (len $candidates) 1) -}} {{- else if $resource -}} {{/* Preserve processable metadata for formats such as SVG and GIF. */}} {{- else -}} {{/* Leave remote or unresolved sources unchanged. */}} {{- end -}}
